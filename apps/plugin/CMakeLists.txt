cmake_minimum_required(VERSION 3.22)
project(VisuSampler VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(VISU_BUILD_PLUGIN "Build the VST3/AU plugin (needs JUCE, fetched automatically)" OFF)
option(VISU_BUILD_TESTS "Build the engine tests" ON)

add_library(visu_engine STATIC engine/Vsamp.cpp engine/Inflate.cpp engine/Library.cpp engine/Engine.cpp)
target_include_directories(visu_engine PUBLIC engine)
set_target_properties(visu_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(visu_engine PRIVATE -Wall -Wextra)
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(visu_engine PUBLIC Threads::Threads)
endif()

if(VISU_BUILD_TESTS)
  enable_testing()
  add_executable(engine_test tests/engine_test.cpp)
  target_link_libraries(engine_test PRIVATE visu_engine)
  target_compile_definitions(engine_test PRIVATE TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/golden" EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../examples")
  add_test(NAME engine COMMAND engine_test)
endif()

if(VISU_BUILD_PLUGIN)
  add_subdirectory(plugin)
endif()
